# Export Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add CSV/JSON/Anki-CSV export for a single deck or all decks, plus make the entire deck card clickable in the home page list.

**Architecture:** Backend adds two JWT-protected GET endpoints that serialize cards and stream a file response. Frontend adds a reusable `CardExportButton` dropdown component wired into DeckPage (single deck) and HomePage (all decks). The stretched-link CSS pattern makes deck cards fully clickable without breaking the delete button.

**Tech Stack:** Go/Gin (backend), Vue 3 + Tailwind + FSD (frontend)

  ```go
  // ExportDeck returns all cards in one deck after verifying ownership.
  func (s *CardService) ExportDeck(ownerID, deckID string) (*DeckExport, error) {
  	deck, err := s.decks.FindByID(deckID)
  	if err != nil {
  		return nil, err
  	}
  	if deck.OwnerID != ownerID {
  		return nil, domain.ErrForbidden
  	}
  	cards, err := s.cards.ListByDeck(deckID)
  	if err != nil {
  		return nil, err
  	}
  	return &DeckExport{Deck: *deck, Cards: cards}, nil
  }

  // ExportAll returns all decks and their cards for a user.
  func (s *CardService) ExportAll(ownerID string) ([]DeckExport, error) {
  	decks, err := s.decks.ListByOwner(ownerID)
  	if err != nil {
  		return nil, err
  	}
  	out := make([]DeckExport, 0, len(decks))
  	for _, deck := range decks {
  		cards, err := s.cards.ListByDeck(deck.ID)
  		if err != nil {
  			return nil, err
  		}
  		out = append(out, DeckExport{Deck: deck, Cards: cards})
  	}
  	return out, nil
  }
  ```

  ```go
  // exportMeta returns content-type and file extension for a given format.
  func exportMeta(format string) (contentType, ext string, err error) {
  	switch format {
  	case "csv":
  		return "text/csv", "csv", nil
  	case "json":
  		return "application/json", "json", nil
  	case "anki-csv":
  		return "text/plain", "txt", nil
  	default:
  		return "", "", errors.New("unsupported format: use csv, json, or anki-csv")
  	}
  }

  // renderExport serializes DeckExport slices to bytes in the requested format.
  func renderExport(format string, decks []service.DeckExport) ([]byte, error) {
  	switch format {
  	case "csv":
  		return renderCSVExport(decks)
  	case "json":
  		return renderJSONExport(decks)
  	case "anki-csv":
  		return renderAnkiCSVExport(decks)
  	default:
  		return nil, errors.New("unsupported format")
  	}
  }

  func renderCSVExport(decks []service.DeckExport) ([]byte, error) {
  	var buf strings.Builder
  	w := csv.NewWriter(&buf)
  	_ = w.Write([]string{"deck_name", "front", "back", "topic", "level", "source_url", "tags"})
  	for _, de := range decks {
  		for _, c := range de.Cards {
  			row := []string{
  				de.Deck.Title,
  				c.Content.Front,
  				c.Content.Back,
  				derefStr(c.Content.Topic),
  				derefStr(c.Content.Level),
  				derefStr(c.Content.SourceURL),
  				strings.Join(c.Content.Tags, "|"),
  			}
  			_ = w.Write(row)
  		}
  	}
  	w.Flush()
  	return []byte(buf.String()), w.Error()
  }

  func renderJSONExport(decks []service.DeckExport) ([]byte, error) {
  	type cardRow struct {
  		Front     string   `json:"front"`
  		Back      string   `json:"back"`
  		Topic     *string  `json:"topic,omitempty"`
  		Level     *string  `json:"level,omitempty"`
  		SourceURL *string  `json:"sourceUrl,omitempty"`
  		Tags      []string `json:"tags"`
  	}
  	type deckRow struct {
  		Name        string    `json:"name"`
  		Description string    `json:"description"`
  		Cards       []cardRow `json:"cards"`
  	}
  	rows := make([]deckRow, 0, len(decks))
  	for _, de := range decks {
  		cards := make([]cardRow, 0, len(de.Cards))
  		for _, c := range de.Cards {
  			cards = append(cards, cardRow{
  				Front:     c.Content.Front,
  				Back:      c.Content.Back,
  				Topic:     c.Content.Topic,
  				Level:     c.Content.Level,
  				SourceURL: c.Content.SourceURL,
  				Tags:      c.Content.Tags,
  			})
  		}
  		rows = append(rows, deckRow{
  			Name:        de.Deck.Title,
  			Description: de.Deck.Description,
  			Cards:       cards,
  		})
  	}
  	return json.MarshalIndent(rows, "", "  ")
  }

  func renderAnkiCSVExport(decks []service.DeckExport) ([]byte, error) {
  	// Anki tab-separated: front TAB back TAB tags (space-separated), no header.
  	var buf strings.Builder
  	for _, de := range decks {
  		for _, c := range de.Cards {
  			buf.WriteString(c.Content.Front)
  			buf.WriteByte('\t')
  			buf.WriteString(c.Content.Back)
  			buf.WriteByte('\t')
  			buf.WriteString(strings.Join(c.Content.Tags, " "))
  			buf.WriteByte('\n')
  		}
  	}
  	return []byte(buf.String()), nil
  }

  func derefStr(s *string) string {
  	if s == nil {
  		return ""
  	}
  	return *s
  }
  ```

  ```go
  func (h *CardHandler) ExportDeck(c *gin.Context) {
  	format := c.Query("format")
  	contentType, ext, err := exportMeta(format)
  	if err != nil {
  		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
  		return
  	}

  	de, err := h.cards.ExportDeck(c.GetString(middleware.ContextUserID), c.Param("id"))
  	if status, failed := cardErrorStatus(err); failed {
  		c.JSON(status, gin.H{"error": err.Error()})
  		return
  	}

  	data, err := renderExport(format, []service.DeckExport{*de})
  	if err != nil {
  		c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
  		return
  	}

  	filename := "flashcards-" + de.Deck.Title + "." + ext
  	c.Header("Content-Disposition", "attachment; filename="+filename)
  	c.Data(http.StatusOK, contentType, data)
  }

  func (h *CardHandler) ExportAll(c *gin.Context) {
  	format := c.Query("format")
  	contentType, ext, err := exportMeta(format)
  	if err != nil {
  		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
  		return
  	}

  	decks, err := h.cards.ExportAll(c.GetString(middleware.ContextUserID))
  	if status, failed := cardErrorStatus(err); failed {
  		c.JSON(status, gin.H{"error": err.Error()})
  		return
  	}

  	data, err := renderExport(format, decks)
  	if err != nil {
  		c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
  		return
  	}

  	c.Header("Content-Disposition", "attachment; filename=flashcards-export."+ext)
  	c.Data(http.StatusOK, contentType, data)
  }
  ```

  ```typescript
  export async function apiBlobRequest(path: string): Promise<{ blob: Blob; filename: string }> {
    const headers: Record<string, string> = {}
    if (authToken) headers.Authorization = `Bearer ${authToken}`

    const response = await fetch(`${API_BASE_URL}${path}`, { headers })

    if (!response.ok) {
      const data = await response.json().catch(() => null)
      const message =
        data && typeof data.error === 'string'
          ? data.error
          : `Request failed (${response.status})`
      throw new ApiError(response.status, message)
    }

    const contentDisposition = response.headers.get('Content-Disposition') ?? ''
    const match = contentDisposition.match(/filename=(.+)/)
    const filename = match ? match[1].trim() : 'export'

    const blob = await response.blob()
    return { blob, filename }
  }
  ```

  ```vue
  <script setup lang="ts">
  import { ref } from 'vue'
  import { exportCards, type ExportFormat } from '@/entities/card'

  const props = defineProps<{ deckId?: string }>()

  const formats: { value: ExportFormat; label: string }[] = [
    { value: 'csv', label: 'CSV' },
    { value: 'json', label: 'JSON' },
    { value: 'anki-csv', label: 'Anki CSV' },
  ]

  const loading = ref(false)
  const open = ref(false)
  const error = ref<string | null>(null)

  async function download(format: ExportFormat) {
    open.value = false
    loading.value = true
    error.value = null
    try {
      const { blob, filename } = await exportCards(format, props.deckId)
      const url = URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = filename
      a.click()
      URL.revokeObjectURL(url)
    } catch (e) {
      error.value = e instanceof Error ? e.message : 'Export failed'
    } finally {
      loading.value = false
    }
  }
  </script>

  <template>
    <div class="relative">
      <button
        class="rounded-lg border border-slate-600 px-4 py-1.5 text-sm font-medium text-slate-300 hover:border-slate-500 hover:text-slate-100 disabled:opacity-50"
        :disabled="loading"
        @click="open = !open"
      >
        {{ loading ? 'Exporting…' : 'Export' }}
      </button>
      <div
        v-if="open"
        class="absolute right-0 top-full z-10 mt-1 rounded-lg border border-slate-700 bg-slate-800 py-1 shadow-xl"
      >
        <button
          v-for="fmt in formats"
          :key="fmt.value"
          class="block w-full px-4 py-2 text-left text-sm text-slate-200 hover:bg-slate-700"
          @click="download(fmt.value)"
        >
          {{ fmt.label }}
        </button>
      </div>
      <p v-if="error" class="mt-1 text-xs text-red-400">{{ error }}</p>
    </div>
  </template>
  ```
