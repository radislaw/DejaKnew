> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add ru/en i18n to the Vue 3 frontend (vue-i18n v9) and Go backend (go-i18n v2), with browser-detect + manual switcher on frontend and locale stored in DB per user.

**Architecture:** vue-i18n Composition API (legacy: false) in `shared/config/`; locale detection from localStorage → navigator.language → 'en'; `Accept-Language` header injected into every API request; go-i18n v2 bundle on backend with embedded TOML catalogs; `users.locale` column stores preference set at registration.

**Tech Stack:** vue-i18n v9, go-i18n/v2, go-toml/v2 (already in go.mod), GORM AutoMigrate (existing pattern)

```typescript
import { createI18n } from 'vue-i18n'
import en from './locales/en.json'
import ru from './locales/ru.json'

export type MessageSchema = typeof en
export type Locale = 'en' | 'ru'

const SUPPORTED: Locale[] = ['en', 'ru']

function detectLocale(): Locale {
  const saved = localStorage.getItem('locale')
  if (saved === 'ru' || saved === 'en') return saved
  const browser = navigator.language.split('-')[0]
  return (SUPPORTED as string[]).includes(browser) ? (browser as Locale) : 'en'
}

export const i18n = createI18n<[MessageSchema], Locale>({
  legacy: false,
  locale: detectLocale(),
  fallbackLocale: 'en',
  messages: { en, ru },
  pluralRules: {
    ru: (choice: number): number => {
      if (choice === 0) return 2
      const teen = choice > 10 && choice < 20
      const end = choice % 10
      if (!teen && end === 1) return 0
      if (!teen && end >= 2 && end <= 4) return 1
      return 2
    },
  },
})
```

Add `currentLocale` tracking alongside `authToken`:

```typescript
import { API_BASE_URL } from '@/shared/config'

let authToken: string | null = null
let currentLocale: string = 'en'

export function setAuthToken(token: string | null): void {
  authToken = token
}

export function setLocale(locale: string): void {
  currentLocale = locale
}

export class ApiError extends Error {
  constructor(
    public readonly status: number,
    message: string,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}

interface RequestOptions {
  method?: string
  body?: unknown
}

export async function apiRequest<T>(path: string, options: RequestOptions = {}): Promise<T> {
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    'Accept-Language': currentLocale,
  }
  if (authToken) headers.Authorization = `Bearer ${authToken}`

  const response = await fetch(`${API_BASE_URL}${path}`, {
    method: options.method ?? 'GET',
    headers,
    body: options.body !== undefined ? JSON.stringify(options.body) : undefined,
  })

  if (response.status === 204) return undefined as T

  const data = await response.json().catch(() => null)
  if (!response.ok) {
    const message =
      data && typeof data.error === 'string'
        ? data.error
        : `Request failed (${response.status})`
    throw new ApiError(response.status, message)
  }
  return data as T
}

export async function apiBlobRequest(path: string): Promise<{ blob: Blob; filename: string }> {
  const headers: Record<string, string> = { 'Accept-Language': currentLocale }
  if (authToken) headers.Authorization = `Bearer ${authToken}`

  const response = await fetch(`${API_BASE_URL}${path}`, { headers })

  if (!response.ok) {
    const data = await response.json().catch(() => null)
    const message =
      data && typeof data.error === 'string'
        ? data.error
        : `Request failed (${response.status})`
    throw new ApiError(response.status, message)
  }

  const contentDisposition = response.headers.get('Content-Disposition') ?? ''
  const match = contentDisposition.match(/filename=(.+)/)
  const filename = match ? match[1].trim() : 'export'

  const blob = await response.blob()
  return { blob, filename }
}
```

```vue
import { useI18n } from 'vue-i18n'
import { setLocale } from '@/shared/api'
import type { Locale } from '@/shared/config/i18n'

const { locale } = useI18n()

function switchLocale() {
  const next: Locale = locale.value === 'ru' ? 'en' : 'ru'
  locale.value = next
  localStorage.setItem('locale', next)
  setLocale(next)
}
</script>
```

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { useI18n } from 'vue-i18n'
import { importCards } from '@/entities/card/api/card.api'
import type { Card } from '@/entities/card'

const { t } = useI18n()

const props = defineProps<{ deckId: string }>()

const emit = defineEmits<{
  imported: [cards: Card[]]
  cancel: []
}>()

const format = ref<'json' | 'csv'>('json')
const fileText = ref('')
const fileName = ref('')
const preview = ref<number | null>(null)
const error = ref<string | null>(null)
const loading = ref(false)

function onFileChange(e: Event) {
  const input = e.target as HTMLInputElement
  const file = input.files?.[0]
  if (!file) return

  fileName.value = file.name
  error.value = null
  preview.value = null

  if (file.name.endsWith('.csv')) format.value = 'csv'
  else if (file.name.endsWith('.json')) format.value = 'json'

  const reader = new FileReader()
  reader.onload = (ev) => {
    fileText.value = (ev.target?.result as string) ?? ''
    previewCount()
  }
  reader.readAsText(file)
}

function previewCount() {
  if (!fileText.value) return
  try {
    if (format.value === 'json') {
      const parsed = JSON.parse(fileText.value)
      preview.value = Array.isArray(parsed) ? parsed.filter((r) => r.front && r.back).length : 0
    } else {
      const lines = fileText.value.split('\n').filter((l) => l.trim())
      preview.value = Math.max(0, lines.length - 1)
    }
    error.value = null
  } catch {
    error.value = t('cardImport.failedToParse')
    preview.value = null
  }
}

async function onSubmit() {
  if (!fileText.value) return
  error.value = null
  loading.value = true
  try {
    const result = await importCards(props.deckId, format.value, fileText.value)
    emit('imported', result.cards)
  } catch (e) {
    error.value = e instanceof Error ? e.message : t('cardImport.importFailed')
  } finally {
    loading.value = false
  }
}
</script>

<template>
  <div class="rounded-xl border border-slate-700 bg-slate-800/60 p-4 flex flex-col gap-3">
    <div class="flex items-center justify-between">
      <span class="text-sm font-medium text-slate-200">{{ t('cardImport.title') }}</span>
      <div class="flex gap-1 rounded-lg bg-slate-900 p-0.5">
        <button
          v-for="f in ['json', 'csv'] as const"
          :key="f"
          class="rounded-md px-3 py-1 text-xs font-medium transition-colors"
          :class="format === f ? 'bg-sky-600 text-white' : 'text-slate-400 hover:text-slate-200'"
          @click="format = f; previewCount()"
        >
          {{ f.toUpperCase() }}
        </button>
      </div>
    </div>

    <label
      class="flex cursor-pointer flex-col items-center gap-2 rounded-lg border border-dashed border-slate-600 px-4 py-6 text-sm text-slate-400 hover:border-slate-500 hover:text-slate-300 transition-colors"
    >
      <span class="text-2xl">📂</span>
      <span>{{ fileName || t('cardImport.chooseFile') }}</span>
```

```typescript
import { defineStore } from 'pinia'
import { computed, ref } from 'vue'
import { setAuthToken } from '@/shared/api'
import type { User } from '@/entities/user'
import { loginRequest, meRequest, registerRequest } from '../api/auth.api'

const TOKEN_KEY = 'flashcards.token'

export const useAuthStore = defineStore('auth', () => {
  const token = ref<string | null>(localStorage.getItem(TOKEN_KEY))
  const user = ref<User | null>(null)

  const isAuthenticated = computed(() => token.value !== null)

  setAuthToken(token.value)

  function applyToken(value: string | null): void {
    token.value = value
    setAuthToken(value)
    if (value) localStorage.setItem(TOKEN_KEY, value)
    else localStorage.removeItem(TOKEN_KEY)
  }

  async function login(email: string, password: string): Promise<void> {
    const res = await loginRequest(email, password)
    applyToken(res.token)
    user.value = res.user
  }

  async function register(email: string, password: string): Promise<void> {
    const locale = localStorage.getItem('locale') ?? 'en'
    const res = await registerRequest(email, password, locale)
    applyToken(res.token)
    user.value = res.user
  }

  async function restore(): Promise<void> {
    if (!token.value) return
    try {
      user.value = await meRequest()
    } catch {
      applyToken(null)
    }
  }

  function logout(): void {
    applyToken(null)
    user.value = null
  }

  return { token, user, isAuthenticated, login, register, restore, logout }
})
```

```go
// Init loads both locale files into the bundle. Call once at startup.

func NewLocalizer(lang string) *goI18n.Localizer {
	return goI18n.NewLocalizer(bundle, lang, "en")
```

```go
// Register creates a new account and returns a signed token.
func (s *AuthService) Register(email, password, locale string) (string, *domain.User, error) {
	existing, err := s.users.FindByEmail(email)
	if err != nil && !errors.Is(err, domain.ErrNotFound) {
		return "", nil, err
	}
	if existing != nil {
		return "", nil, domain.ErrUserExists
	}

	hash, err := auth.HashPassword(password)
	if err != nil {
		return "", nil, err
	}

	if locale != "ru" && locale != "en" {
		locale = "en"
	}

	user := &domain.User{
		ID:           uuid.NewString(),
		Email:        email,
		PasswordHash: hash,
		Locale:       locale,
	}
	if err := s.users.Create(user); err != nil {
		return "", nil, err
	}

	token, err := s.tokens.Issue(user.ID)
	if err != nil {
		return "", nil, err
	}
	return token, user, nil
}
```

```go
package middleware

import (
	"strings"

	"github.com/gin-gonic/gin"
)

const ContextLocale = "locale"

var supportedLocales = map[string]bool{"en": true, "ru": true}

// Locale extracts the primary language tag from Accept-Language header
// and injects it into the Gin context as "locale". Falls back to "en".
func Locale() gin.HandlerFunc {
	return func(c *gin.Context) {
		lang := parseAcceptLanguage(c.GetHeader("Accept-Language"))
		c.Set(ContextLocale, lang)
		c.Next()
	}
}

func parseAcceptLanguage(header string) string {
	if header == "" {
		return "en"
	}
	// Take first language tag, strip region subtag (e.g. "ru-RU" → "ru")
	first := strings.Split(header, ",")[0]
	first = strings.Split(first, ";")[0]
	first = strings.Split(strings.TrimSpace(first), "-")[0]
	first = strings.ToLower(strings.TrimSpace(first))
	if supportedLocales[first] {
		return first
	}
	return "en"
}
```

```go
package handler

import (
	"net/http"
	"time"

	"flashcards/api/internal/auth"
	"flashcards/api/internal/middleware"

	"github.com/gin-contrib/cors"
	"github.com/gin-gonic/gin"
)

func Router(tokens *auth.TokenManager, authHandler *AuthHandler, deckHandler *DeckHandler, cardHandler *CardHandler, studyHandler *StudyHandler) *gin.Engine {
	r := gin.Default()

	r.Use(cors.New(cors.Config{
		AllowOrigins:     []string{"http://localhost:5173", "http://localhost:5174"},
		AllowMethods:     []string{"GET", "POST", "PATCH", "DELETE", "OPTIONS"},
		AllowHeaders:     []string{"Origin", "Content-Type", "Authorization", "Accept-Language"},
		ExposeHeaders:    []string{"Content-Length", "Content-Disposition"},
		AllowCredentials: false,
		MaxAge:           12 * time.Hour,
	}))

	r.Use(middleware.Locale())

	r.GET("/health", func(c *gin.Context) {
		c.JSON(http.StatusOK, gin.H{"status": "ok"})
	})

	api := r.Group("/api/v1")
	{
		api.POST("/auth/register", authHandler.Register)
		api.POST("/auth/login", authHandler.Login)

		protected := api.Group("")
		protected.Use(middleware.RequireAuth(tokens))
		{
			protected.GET("/me", authHandler.Me)

			protected.POST("/decks", deckHandler.Create)
			protected.GET("/decks", deckHandler.List)
			protected.GET("/decks/:id", deckHandler.Get)
			protected.PATCH("/decks/:id", deckHandler.Update)
			protected.DELETE("/decks/:id", deckHandler.Delete)

			protected.POST("/decks/:id/cards", cardHandler.Create)
			protected.POST("/decks/:id/import", cardHandler.Import)
			protected.GET("/decks/:id/export", cardHandler.ExportDeck)
			protected.GET("/export", cardHandler.ExportAll)
			protected.PATCH("/cards/:id", cardHandler.Update)
			protected.DELETE("/cards/:id", cardHandler.Delete)

			protected.GET("/study/:deckId/queue", studyHandler.GetQueue)
			protected.POST("/study/:cardId/review", studyHandler.SubmitReview)
		}
	}

	return r
}
```
